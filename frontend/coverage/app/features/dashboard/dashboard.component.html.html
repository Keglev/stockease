<!-- Four headline figures, then the two report visualizations. No chart markup lives here: both
     visualizations are card components that take finished derivations, so this file is the KPI row
     and the layout around it and nothing else. The low-stock figure is a button rather than a
     figure because its detail opens in a dialog instead of a section of its own. -->
<div class="dashboard-toolbar">
  <h1>{{ 'dashboard.title' | translate }}</h1>
  <button matButton="outlined" type="button" class="dashboard-refresh" (click)="refresh()">
    {{ 'dashboard.refresh' | translate }}
  </button>
</div>
 
<!-- Rendered verbatim: this text comes from the backend envelope, which has no i18n. -->
@if (error(); as message) {
  <p class="dashboard-error" role="alert">{{ message }}</p>
}
 
<!-- While the load failed, every KPI shows an em dash. A zero here is indistinguishable from a real
     zero, and the signals still hold their defaults, so rendering them would present "no data" as
     data - the exact reading a user takes from a dashboard full of zeros. -->
<div class="kpi-row">
  <mat-card class="kpi-card kpi-products">
    <mat-card-content>
      <p class="kpi-label">{{ 'dashboard.kpi.products' | translate }}</p>
      <p class="kpi-value">{{ error() ? '—' : totalProducts() }}</p>
    </mat-card-content>
  </mat-card>
 
  <!-- A real button, not a card with a click handler: the list behind it has to be reachable by
       keyboard and announced as an action, which only button semantics give for free. Inert under
       error, because the em dash means the rows were never loaded - there is nothing to open. -->
  <mat-card class="kpi-card kpi-low-stock" [class.kpi-actionable]="!error()">
    <mat-card-content>
      @if (error()) {
        <p class="kpi-label">{{ 'dashboard.kpi.lowStock' | translate }}</p>
        <p class="kpi-value">—</p>
      } @else {
        <button
          type="button"
          class="kpi-button low-stock-open"
          [attr.aria-label]="'dashboard.kpi.lowStockOpen' | translate"
          (click)="openLowStock()"
        >
          <span class="kpi-label">{{ 'dashboard.kpi.lowStock' | translate }}</span>
          <span class="kpi-value">{{ lowStock().length }}</span>
        </button>
      }
    </mat-card-content>
  </mat-card>
 
  <mat-card class="kpi-card kpi-overdue">
    <mat-card-content>
      <p class="kpi-label">{{ 'dashboard.kpi.overdue' | translate }}</p>
      <p class="kpi-value">{{ error() ? '—' : overdueCount() }}</p>
    </mat-card-content>
  </mat-card>
 
  <mat-card class="kpi-card kpi-gross-profit">
    <mat-card-content>
      <p class="kpi-label">{{ 'dashboard.kpi.grossProfit' | translate }}</p>
      <p
        class="kpi-value"
        [class.kpi-negative]="!error() && grossProfit() < 0"
      >
        {{ error() ? '—' : (grossProfit() | appCurrency) }}
      </p>
    </mat-card-content>
  </mat-card>
</div>
