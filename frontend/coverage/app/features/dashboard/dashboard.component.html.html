<!-- Four headline figures, then the two report visualizations. No chart markup lives here: both
     visualizations are card components that take finished derivations, so this file is the KPI row
     and the layout around it and nothing else. The low-stock figure is a button rather than a
     figure because its detail opens in a dialog instead of a section of its own. -->
<div class="dashboard-toolbar">
  <h1>{{ 'dashboard.title' | translate }}</h1>
  <button matButton="outlined" type="button" class="dashboard-refresh" (click)="refresh()">
    {{ 'dashboard.refresh' | translate }}
  </button>
</div>
 
<!-- Rendered verbatim: this text comes from the backend envelope, which has no i18n. -->
@if (error(); as message) {
  <p class="dashboard-error" role="alert">{{ message }}</p>
}
 
<!-- Three states per figure, in the order they can occur: the request is out, the request lost, the
     request answered. The placeholder comes first because during the load the other two are both
     lies - the signals still hold their zero defaults, and a zero here is indistinguishable from a
     real zero, which presents "not yet known" as data. The dash is the failed load and stays
     exactly as it was; what it no longer has to cover is the wait.
 
     aria-hidden on the placeholder and aria-busy on the card: the shape is decoration and has
     nothing to read out, while the card announces that its content is still being fetched. -->
<div class="kpi-row">
  <mat-card class="kpi-card kpi-products" [attr.aria-busy]="loadingProducts() ? 'true' : null">
    <mat-card-content>
      <p class="kpi-label">{{ 'dashboard.kpi.products' | translate }}</p>
      <p class="kpi-value">
        @if (loadingProducts()) {
          <span class="skeleton skeleton-kpi" aria-hidden="true"></span>
        } @else if (error()) {
          —
        } @else {
          {{ totalProducts() }}
        }
      </p>
    </mat-card-content>
  </mat-card>
 
  <!-- A real button, not a card with a click handler: the list behind it has to be reachable by
       keyboard and announced as an action, which only button semantics give for free. Inert under
       error, because the em dash means the rows were never loaded - there is nothing to open. The
       load branches the same way and for the same reason: the rows have not arrived yet, so the
       control that opens them is not offered rather than offered over an empty list. -->
  <mat-card
    class="kpi-card kpi-low-stock"
    [class.kpi-actionable]="!error() && !loadingLowStock()"
    [attr.aria-busy]="loadingLowStock() ? 'true' : null"
  >
    <mat-card-content>
      @if (loadingLowStock()) {
        <p class="kpi-label">{{ 'dashboard.kpi.lowStock' | translate }}</p>
        <p class="kpi-value"><span class="skeleton skeleton-kpi" aria-hidden="true"></span></p>
      } @else if (error()) {
        <p class="kpi-label">{{ 'dashboard.kpi.lowStock' | translate }}</p>
        <p class="kpi-value">—</p>
      } @else {
        <button
          type="button"
          class="kpi-button low-stock-open"
          [attr.aria-label]="'dashboard.kpi.lowStockOpen' | translate"
          (click)="openLowStock()"
        >
          <span class="kpi-label">{{ 'dashboard.kpi.lowStock' | translate }}</span>
          <span class="kpi-value">{{ lowStock().length }}</span>
        </button>
      }
    </mat-card-content>
  </mat-card>
 
  <mat-card class="kpi-card kpi-overdue" [attr.aria-busy]="loadingOverdue() ? 'true' : null">
    <mat-card-content>
      <p class="kpi-label">{{ 'dashboard.kpi.overdue' | translate }}</p>
      <p class="kpi-value">
        @if (loadingOverdue()) {
          <span class="skeleton skeleton-kpi" aria-hidden="true"></span>
        } @else if (error()) {
          —
        } @else {
          {{ overdueCount() }}
        }
      </p>
    </mat-card-content>
  </mat-card>
 
  <mat-card class="kpi-card kpi-gross-profit" [attr.aria-busy]="loadingProfit() ? 'true' : null">
    <mat-card-content>

  </mat-card>
</div>
